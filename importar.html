Importar Inventario
Importar stock vencido desde Vencidos a 01-2025 para Metanoia.xlsx
1
Correr el siguiente SQL en Supabase → SQL Editor para crear la tabla: 
CREATE TABLE inventario (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  codigo text,
  nombre text NOT NULL,
  fabricante text,
  categoria text,
  partida text,
  deposito text,
  stock_actual integer DEFAULT 0,
  stock_reservado integer DEFAULT 0,
  stock_disponible integer DEFAULT 0,
  fecha_vencimiento date,
  observaciones text,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);
ALTER TABLE inventario ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all" ON inventario FOR ALL USING (true) WITH CHECK (true);
